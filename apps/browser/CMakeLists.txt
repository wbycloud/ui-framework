cmake_minimum_required(VERSION 3.20)
if(NOT TARGET ui_framework_shared)
    project(restricted_browser VERSION 0.3.0 LANGUAGES C)
    if(MSVC AND CMAKE_CL_SHOWINCLUDES_PREFIX MATCHES "娉")
        set(CMAKE_CL_SHOWINCLUDES_PREFIX "注意: 包含文件:")
    endif()
    set(UI_BROWSER_SDK_DIR "" CACHE PATH "Matching extracted SDK; builds app without framework sources")
    if(UI_BROWSER_SDK_DIR)
        add_library(ui_framework_shared SHARED IMPORTED GLOBAL)
        set_target_properties(ui_framework_shared PROPERTIES IMPORTED_LOCATION "${UI_BROWSER_SDK_DIR}/bin/ui_framework.dll"
            IMPORTED_IMPLIB "${UI_BROWSER_SDK_DIR}/lib/ui_framework_runtime.lib"
            INTERFACE_INCLUDE_DIRECTORIES "${UI_BROWSER_SDK_DIR}/include"
            INTERFACE_COMPILE_DEFINITIONS UI_FRAMEWORK_BUILD_SHARED)
        add_library(lexbor_static STATIC IMPORTED GLOBAL)
        set_target_properties(lexbor_static PROPERTIES IMPORTED_LOCATION "${UI_BROWSER_SDK_DIR}/third_party/lexbor/lib/lexbor_static.lib"
            INTERFACE_COMPILE_DEFINITIONS LEXBOR_STATIC)
        set(BROWSER_LEXBOR_INCLUDE "${UI_BROWSER_SDK_DIR}/third_party/lexbor/include")
        add_executable(uapp_pack IMPORTED GLOBAL)
        set_target_properties(uapp_pack PROPERTIES IMPORTED_LOCATION "${UI_BROWSER_SDK_DIR}/bin/uapp_pack.exe")
    else()
        set(UI_FRAMEWORK_SOURCE_DIR "D:/应用软件框架/应用层序框架" CACHE PATH "Matching framework source")
        set(UI_BROWSER_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}" CACHE PATH "Built-in browser source" FORCE)
        add_subdirectory("${UI_FRAMEWORK_SOURCE_DIR}" framework)
        return()
    endif()
endif()

set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
option(BROWSER_BUILD_CHECKS "Build browser public C verification programs" OFF)
option(BROWSER_DIAGNOSTIC_SYMBOLS "Include symbols for local crash diagnosis" OFF)
if(NOT BROWSER_LEXBOR_INCLUDE)
    set(BROWSER_LEXBOR_INCLUDE "${UI_LEXBOR_SOURCE_DIR}/source")
endif()
if(NOT UI_FRAMEWORK_SOURCE_DIR)
    get_target_property(UI_FRAMEWORK_SOURCE_DIR ui_framework_shared SOURCE_DIR)
endif()
if(BROWSER_DIAGNOSTIC_SYMBOLS)
    target_compile_options(ui_framework_shared PRIVATE /Z7)
    target_link_options(ui_framework_shared PRIVATE /DEBUG)
endif()
add_library(browser_app MODULE src/browser_app.c src/browser_net.c src/browser_url.c src/browser_json.c src/browser_encoding.c src/browser_image.c src/browser_settings.c src/browser_diagnostic.c)
target_include_directories(browser_app PRIVATE src "${BROWSER_LEXBOR_INCLUDE}")
target_link_libraries(browser_app PRIVATE ui_framework_shared lexbor_static winhttp windowscodecs ole32 uuid comctl32)
target_compile_definitions(browser_app PRIVATE _CRT_SECURE_NO_WARNINGS)
if(MSVC)
    target_compile_options(browser_app PRIVATE /W4 /WX /utf-8)
    if(BROWSER_DIAGNOSTIC_SYMBOLS)
        target_compile_options(browser_app PRIVATE /Z7)
        target_link_options(browser_app PRIVATE /DEBUG)
    endif()
endif()
file(GLOB browser_resources CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/resources/*")
if(TARGET framework_host)
    set(browser_package "${CMAKE_CURRENT_BINARY_DIR}/../browser.uapp")
else()
    set(browser_package "${CMAKE_CURRENT_BINARY_DIR}/browser.uapp")
endif()
add_custom_command(OUTPUT "${browser_package}"
    COMMAND ${CMAKE_COMMAND} -E make_directory "${CMAKE_CURRENT_BINARY_DIR}/package"
    COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_CURRENT_SOURCE_DIR}/resources" "${CMAKE_CURRENT_BINARY_DIR}/package"
    COMMAND ${CMAKE_COMMAND} -E copy "$<TARGET_FILE:browser_app>" "${CMAKE_CURRENT_BINARY_DIR}/package/browser_app.dll"
    COMMAND $<TARGET_FILE:uapp_pack> "${CMAKE_CURRENT_SOURCE_DIR}/resources/app.ini" "${CMAKE_CURRENT_BINARY_DIR}/package" "${browser_package}"
    DEPENDS browser_app uapp_pack ${browser_resources} VERBATIM)
add_custom_target(browser_package ALL DEPENDS "${browser_package}")
if(TARGET framework_host)
    add_dependencies(framework_host browser_package)
endif()
if(BROWSER_BUILD_CHECKS)
    add_executable(browser_m2_check tests/m2_public.c src/browser_json.c)

    target_include_directories(browser_url_check PRIVATE src "${BROWSER_LEXBOR_INCLUDE}")

    if(MSVC)
        foreach(check IN ITEMS browser_document_check browser_workspace_check browser_desktop_check browser_failure_check browser_url_check browser_settings_check browser_m01_check browser_reading_check browser_encoding_check browser_image_check browser_m2_check)
